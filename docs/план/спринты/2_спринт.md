# 1) Патч: `gateway/ngnix.conf` — rate-limit ТОЛЬКО для **POST /tasks**

Вставь пометки `+++` в свои места.

```nginx
worker_processes 1;
events { worker_connections 1024; }

http {
  include       mime.types;
  default_type  application/octet-stream;
  sendfile      on;
  keepalive_timeout  65;

  # +++ RATE-LIMIT только для POST /tasks (ключ — IP)
  map $request_method$uri $tasks_post_key {
    default "";
    ~^POST/tasks/?$ $binary_remote_addr;
  }
  limit_req_zone $tasks_post_key zone=task_post:10m rate=10r/s;

  upstream auth_upstream  { server auth-svc:8000; }
  upstream task_upstream  { server task-svc:8000; }
  upstream notify_upstream{ server notify-svc:8000; }

  server {
    listen 80;

    location = /healthz { return 200 "ok\n"; }

    location /auth/ {
      proxy_pass http://auth_upstream/;
      proxy_set_header Host $host;
      proxy_set_header X-Real-IP $remote_addr;
      proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
      proxy_read_timeout 30s;
      # +++ полезно для трассировки
      proxy_set_header X-Request-ID $request_id;
    }

    # +++ ограничиваем только POST /tasks и /tasks/
    location = /tasks {
      if ($tasks_post_key) { limit_req zone=task_post burst=20 nodelay; }
      proxy_pass http://task_upstream/;
      proxy_set_header Host $host;
      proxy_set_header X-Real-IP $remote_addr;
      proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
      proxy_set_header X-Request-ID $request_id;

    }
    location = /tasks/ {

      proxy_set_header Host $host;
      proxy_set_header X-Real-IP $remote_addr;
      proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
```

```python
from __future__ import annotations
from datetime import datetime
from typing import Optional, Literal
from pydantic import BaseModel, Field

TaskStatus = Literal["todo", "in_progress", "done", "archived"]


class TaskCreate(BaseModel):
    """Вход: создание задачи."""
    title: str = Field(min_length=1, max_length=255)
    description: Optional[str] = None
    priority: int = Field(default=3, ge=1, le=5)
    due_at: Optional[datetime] = None


class TaskUpdate(BaseModel):
    """Вход: частичное обновление задачи."""
    title: Optional[str] = Field(default=None, min_length=1, max_length=255)
    description: Optional[str] = None
    priority: Optional[int] = Field(default=None, ge=1, le=5)
    due_at: Optional[datetime] = None
    status: Optional[TaskStatus] = None


class TaskOut(BaseModel):
    """Выход: задача."""
    id: str
    owner_id: str
    title: str
    description: Optional[str]
    status: TaskStatus
    priority: int
    due_at: Optional[datetime]
    created_at: datetime
    updated_at: datetime
```

## 4.5 `task-svc/app/deps.py` — JWT-проверка (без требования поля `type`)

```python
from __future__ import annotations
from pathlib import Path
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import jwt
from app.settings import settings

bearer = HTTPBearer(auto_error=True)
PUBLIC_KEY = Path(settings.JWT_PUBLIC_KEY_PATH).read_text()
ALGO = "RS256"


def _decode_access(token: str) -> dict:
    """Декод JWT без audience-проверки, но с проверкой подписи/срока."""
    return jwt.decode(token, PUBLIC_KEY, algorithms=[ALGO], options={"verify_aud": False})


async def get_current_user_id(creds: HTTPAuthorizationCredentials = Depends(bearer)) -> str:
    """
    DI: возвращает user_id (sub) из access-JWT.
    """
    try:
        payload = _decode_access(creds.credentials)
        if payload.get("iss") != settings.JWT_ISS:
            raise ValueError("bad issuer")
        sub = payload.get("sub")
        if not sub:
            raise ValueError("no sub")
        return sub
    except Exception as _:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="invalid token") from None
```

## 4.6 `task-svc/app/routers/tasks.py`

```python
from __future__ import annotations
import uuid
from datetime import datetime
from typing import Annotated, Optional
from fastapi import APIRouter, Depends, HTTPException, Query, status, BackgroundTasks
from sqlalchemy import select, func, and_
from sqlalchemy.ext.asyncio import AsyncSession
from app.db import get_session
from app.models import Task
from app.schemes import TaskCreate, TaskUpdate, TaskOut, PaginatedTasks, BulkStatusIn
from app.deps import get_current_user_id
from app.settings import settings
import httpx

router = APIRouter(prefix="/tasks", tags=["tasks"])


def _to_out(t: Task) -> TaskOut:
    return TaskOut(
        id=t.id, owner_id=t.owner_id, title=t.title, description=t.description,
        status=t.status, priority=t.priority, due_at=t.due_at,
        created_at=t.created_at, updated_at=t.updated_at
    )


async def _notify_task_done(task_id: str, owner_id: str) -> None:
    payload = {
        "type": "task_done",
        "task_id": task_id,
        "user_id": owner_id,
        "ts": datetime.utcnow().isoformat() + "Z",
    }
    url = settings.NOTIFY_SVC_URL.rstrip("/") + "/events"
    timeout = httpx.Timeout(5.0, connect=2.0)
    async with httpx.AsyncClient(timeout=timeout) as client:
        try:
            await client.post(url, json=payload)
        except Exception:
            # Нотификация "best effort" — не валим основную бизнес-операцию
            pass


@router.post("", response_model=TaskOut, status_code=status.HTTP_201_CREATED)
async def create_task(
    data: TaskCreate,
    user_id: Annotated[str, Depends(get_current_user_id)],
    session: Annotated[AsyncSession, Depends(get_session)],
) -> TaskOut:
    task = Task(
        id=str(uuid.uuid4()),
        owner_id=user_id,
        title=data.title,
        description=data.description,
        priority=data.priority,
        due_at=data.due_at,
    )
    session.add(task)
    await session.commit()
    await session.refresh(task)
    return _to_out(task)


@router.get("", response_model=PaginatedTasks)
async def list_tasks(
    user_id: Annotated[str, Depends(get_current_user_id)],
    session: Annotated[AsyncSession, Depends(get_session)],
    status_f: Optional[str] = Query(default=None, alias="status"),
    q: Optional[str] = None,
    min_priority: Optional[int] = Query(default=None, ge=1, le=5),
    max_priority: Optional[int] = Query(default=None, ge=1, le=5),
    due_from: Optional[datetime] = None,
    due_to: Optional[datetime] = None,
    limit: int = Query(default=20, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
) -> PaginatedTasks:
    cond = [Task.owner_id == user_id]
    if status_f:
        cond.append(Task.status == status_f)
    if q:
        like = f"%{q}%"
        cond.append((Task.title.ilike(like)) | (Task.description.ilike(like)))
    if min_priority is not None:
        cond.append(Task.priority >= min_priority)
    if max_priority is not None:
        cond.append(Task.priority <= max_priority)
    if due_from:
        cond.append(and_(Task.due_at.is_not(None), Task.due_at >= due_from))
    if due_to:
        cond.append(and_(Task.due_at.is_not(None), Task.due_at <= due_to))

    total = (await session.execute(select(func.count()).select_from(Task).where(*cond))).scalar_one()
    rows = (await session.execute(
        select(Task).where(*cond).order_by(Task.created_at.desc()).limit(limit).offset(offset)
    )).scalars().all()

    return PaginatedTasks(items=[_to_out(t) for t in rows], total=total, limit=limit, offset=offset)


@router.get("/{task_id}", response_model=TaskOut)
async def get_task(
    task_id: str,
    user_id: Annotated[str, Depends(get_current_user_id)],
    session: Annotated[AsyncSession, Depends(get_session)],
) -> TaskOut:
    t = (await session.execute(select(Task).where(Task.id == task_id, Task.owner_id == user_id))).scalar_one_or_none()
    if not t:
        raise HTTPException(status_code=404, detail="task not found")
    return _to_out(t)


@router.patch("/{task_id}", response_model=TaskOut)
async def update_task(
    task_id: str,
    data: TaskUpdate,
    bg: BackgroundTasks,
    user_id: Annotated[str, Depends(get_current_user_id)],
    session: Annotated[AsyncSession, Depends(get_session)],
) -> TaskOut:
    t = (await session.execute(select(Task).where(Task.id == task_id, Task.owner_id == user_id))).scalar_one_or_none()
    if not t:
        raise HTTPException(status_code=404, detail="task not found")
    prev_status = t.status

    for field, value in data.model_dump(exclude_unset=True).items():
        setattr(t, field, value)

    await session.commit()
    await session.refresh(t)

    if prev_status != "done" and t.status == "done":
        bg.add_task(_notify_task_done, t.id, t.owner_id)

    return _to_out(t)


@router.post("/bulk-status", response_model=int)
async def bulk_status(
    payload: BulkStatusIn,
    bg: BackgroundTasks,
    user_id: Annotated[str, Depends(get_current_user_id)],
    session: Annotated[AsyncSession, Depends(get_session)],
) -> int:
    rows = (await session.execute(
        select(Task).where(Task.owner_id == user_id, Task.id.in_(payload.ids))
    )).scalars().all()
    if not rows:
        return 0

    changed = 0
    for t in rows:
        prev = t.status
        t.status = payload.status
        if prev != "done" and payload.status == "done":
            bg.add_task(_notify_task_done, t.id, t.owner_id)
        changed += 1

    await session.commit()
    return changed
```

```python
from __future__ import annotations
import asyncio
from logging.config import fileConfig
from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import AsyncEngine, create_async_engine
from alembic import context

from app.db import Base
from app import models  # noqa: F401
from app.settings import settings

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


async def run_migrations_online() -> None:
    connectable: AsyncEngine = create_async_engine(settings.DATABASE_URL, poolclass=pool.NullPool)
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata, compare_type=True)
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_offline() -> None:
    context.configure(url=settings.DATABASE_URL, target_metadata=target_metadata, literal_binds=True, compare_type=True)
    with context.begin_transaction():
        context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    asyncio.run(run_migrations_online())
```
